# Layer 04 (debug line arch remainder, arm64: Hi3798MV200H).
#
# Recipe: alldefconfig base + 01-arch-arm64.txt + 02-select-y.txt +
# 03-debug-common.txt + this file -> olddefconfig. This file carries the
# arm64-only boot gates and the per-arch PMU symbol; everything
# architecture-common lives in 03-debug-common.txt.
# Build targets: Image + board dtbs; boot over tftp, debug through the serial
# console, /dev/mem and debugfs.
#
# Writing rule: comments on their own lines; CONFIG lines never carry trailing
# inline comments (merge/verification match whole lines).

# --- PSCI boot (the board dts sets enable-method=psci; SMP needs it for the
#     secondary cores) ---
CONFIG_ARM_PSCI_FW=y
CONFIG_ARM_GIC=y
CONFIG_ARM_ARCH_TIMER=y

# --- optimization posture ---
# LD_DEAD_CODE_DATA_ELIMINATION is not spelled out here: arm64 has no
# HAVE_LD_DEAD_CODE_DATA_ELIMINATION, the symbol is invisible and "off" holds
# by absence (measured: the arm32 equivalent had to be deselected explicitly
# -- see 04-debug-arm.txt).

# --- dtb-required driver whose alldefconfig default is off ---
# arm,cortex-a53-pmu of the cpus node (perf events; the common core is in
# 03-debug-common.txt).
CONFIG_ARM_PMUV3=y
