# Layer 02 (flavor 1: drivers built in) -- architecture-independent driver
# layer.
#
# Admission rule: every board-level driver claimed by the enabled nodes of the
# three board device trees -- regardless of which SoC the IP belongs to
# (PL08x/hinfc610/SATA/SiI9134 and friends travel in the same binary: the
# unified kernel is one image per architecture, boards differ only by DTB).
# Layer 01 carries boot-critical options only. This file and 02-select-m.txt are
# mutually exclusive:
#   select-y = everything built in (release standard; boots without an
#              initramfs)
#   select-m = modularized (see that file's header)
#
# Skeleton parents: subsystem gates the distribution baseline sets to =y and
# the alldefconfig base used by the debug line leaves off. NETDEVICES is the
# notable trap: its default is `y if UML` only; DMADEVICES and RC_DEVICES have
# no default at all. Without these lines the whole driver families below are
# invisible to Kconfig.
CONFIG_USB=y
CONFIG_USB_EHCI_HCD=y
CONFIG_USB_OHCI_HCD=y
CONFIG_SCSI=y
CONFIG_SOUND=y
CONFIG_SND=y
CONFIG_SND_SOC=y
CONFIG_DRM=y
CONFIG_MMC=y
CONFIG_HID=y
CONFIG_HID_GENERIC=y
CONFIG_USB_HID=y
CONFIG_MEDIA_SUPPORT=y
CONFIG_RC_CORE=y
# Exception: CONFIG_VIDEO_DEV has a default (any MEDIA_*_SUPPORT = y points at
# y), but the distribution baseline pins it to m and olddefconfig preserves any
# existing legal value -- without this line the VIDEO_HISTB_* drivers only
# reach m under the distribution baseline. Needed under both baselines.
CONFIG_VIDEO_DEV=y
CONFIG_I2C=y
CONFIG_NET=y
CONFIG_NETDEVICES=y
CONFIG_ETHERNET=y
CONFIG_NET_VENDOR_HISILICON=y
CONFIG_DMADEVICES=y
